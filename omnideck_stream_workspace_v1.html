use vulkano::pipeline::GraphicsPipeline;
use tokio::sync::mpsc::{channel, Receiver};

// Low latency zero-copy frame loop (NVENC hardware decode)
pub async fn init_ultra_stream(config: StreamConfig) -> Result<FramePipe> {
    let (tx, mut rx) = channel::<H265Packet>(128);
    let pipeline = VulkanDescriptor::builder()
        .enable_hdr10(true)
        .target_framerate(144.0)
        .max_jitter_buffer_us(800)
        .build()?;

    // Precision touch calibration: Raw HID injected into hypervisor
    tokio::spawn(async move {
        pipeline.dispatch_subsecond_events(InputMode::AbsoluteTouch).await;
    });

    Ok(FramePipe::new(tx))
}
